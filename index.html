Identificacao e Metadados do Projeto

Versao 1.0 | 3FN Homologada

1. Escopo do Sistema

O sistema proposto para a KaBuM! gerencia as operacoes centrais de vendas da loja virtual em um banco de dados relacional centralizado, contemplando cadastro e consulta de clientes, gerenciamento de enderecos, catalogo de produtos, historico de precos, controle de estoque, montagem de carrinho, registro de pedidos e logs de auditoria. Os pagamentos sao registrados via processo controlado sem comunicacao direta com gateways financeiros; enderecos nao possuem integracao com transportadoras; e a emissao de documentos fiscais oficiais permanece externa ao escopo.

2. Tecnologia e Padroes

  • SGBD: MySQL 8.0.16+ (Engine InnoDB)
  • Codificacao: UTF-8 (utf8mb4)
  • Padrao de Nomenclatura: snake_case (singular)
  • Conformidade Legal: LGPD (Lei 13.709/2018) e CDC
  • Orientacao Academica: Prof. Dr. Diego Carvalho

CREATE DATABASE IF NOT EXISTS ecommerce;
USE ecommerce;

-- TABELA CLIENTE
CREATE TABLE cliente (
    id_cliente BIGINT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    cpf VARCHAR(14) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL UNIQUE,
    senha_hash VARCHAR(255) NOT NULL,
    telefone VARCHAR(20) NOT NULL,
    consentimento_lgpd TINYINT(1) NOT NULL,
    data_consentimento DATETIME NOT NULL,
    anonimizado TINYINT(1) NOT NULL DEFAULT 0,
    ativo TINYINT(1) NOT NULL DEFAULT 1
) ENGINE=InnoDB;

-- TABELA ENDERECO
CREATE TABLE endereco (
    id_endereco BIGINT AUTO_INCREMENT PRIMARY KEY,
    id_cliente BIGINT NOT NULL,
    cep CHAR(8) NOT NULL,
    pais VARCHAR(50) NOT NULL DEFAULT 'Brasil',
    estado CHAR(2) NOT NULL,
    cidade VARCHAR(100) NOT NULL,
    rua VARCHAR(150) NOT NULL,
    numero VARCHAR(20) NOT NULL,
    complemento VARCHAR(100) NULL,
    tipo_endereco VARCHAR(20) NOT NULL,
    endereco_principal TINYINT(1) NOT NULL DEFAULT 0,
    ativo TINYINT(1) NOT NULL DEFAULT 1,

    CONSTRAINT fk_endereco_cliente
        FOREIGN KEY (id_cliente)
        REFERENCES cliente(id_cliente),

    CHECK (tipo_endereco IN ('Residencial', 'Comercial'))
) ENGINE=InnoDB;

-- TABELA PRODUTO
CREATE TABLE produto (
    id_produto BIGINT AUTO_INCREMENT PRIMARY KEY,
    sku VARCHAR(50) NOT NULL UNIQUE,
    nome VARCHAR(150) NOT NULL,
    preco DECIMAL(10,2) NOT NULL,
    status VARCHAR(20) NOT NULL,
    marca VARCHAR(50) NOT NULL,
    categoria VARCHAR(50) NOT NULL,
    referencia VARCHAR(100) NULL,
    desconto DECIMAL(10,2) NOT NULL DEFAULT 0.00,

    CHECK (preco >= 0),
    CHECK (desconto >= 0 AND desconto <= preco),
    CHECK (status IN ('Ativo', 'Inativo'))
) ENGINE=InnoDB;

-- TABELA HISTORICO_PRECO
CREATE TABLE historico_preco (
    id_historico_preco BIGINT AUTO_INCREMENT PRIMARY KEY,
    id_produto BIGINT NOT NULL,
    preco_anterior DECIMAL(10,2) NOT NULL,
    preco_novo DECIMAL(10,2) NOT NULL,
    desconto_anterior DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    desconto_novo DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    data_alteracao DATETIME NOT NULL,

    CONSTRAINT fk_historico_produto
        FOREIGN KEY (id_produto)
        REFERENCES produto(id_produto),

    CHECK (preco_anterior >= 0),
    CHECK (preco_novo >= 0),
    CHECK (desconto_anterior >= 0),
    CHECK (desconto_novo >= 0)
) ENGINE=InnoDB;

-- TABELA ESTOQUE
CREATE TABLE estoque (
    id_estoque BIGINT AUTO_INCREMENT PRIMARY KEY,
    id_produto BIGINT NOT NULL UNIQUE,
    quantidade_disponivel INT NOT NULL,

    CONSTRAINT fk_estoque_produto
        FOREIGN KEY (id_produto)
        REFERENCES produto(id_produto),

    CHECK (quantidade_disponivel >= 0)
) ENGINE=InnoDB;

-- TABELA PEDIDO
CREATE TABLE pedido (
    id_pedido BIGINT AUTO_INCREMENT PRIMARY KEY,
    id_cliente BIGINT NOT NULL,
    id_endereco BIGINT NOT NULL,
    data_pedido DATETIME NOT NULL,
    data_entrega DATETIME NULL,
    data_cancelamento DATETIME NULL,
    valor_desconto DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    valor_frete DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    subtotal DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    quantidade_itens INT NOT NULL DEFAULT 0,
    valor_total DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    status VARCHAR(20) NOT NULL,
    forma_pagamento VARCHAR(30) NOT NULL,

    CONSTRAINT fk_pedido_cliente
        FOREIGN KEY (id_cliente)
        REFERENCES cliente(id_cliente),

    CONSTRAINT fk_pedido_endereco
        FOREIGN KEY (id_endereco)
        REFERENCES endereco(id_endereco),

    CHECK (valor_desconto >= 0),
    CHECK (valor_frete >= 0),
    CHECK (subtotal >= 0),
    CHECK (quantidade_itens >= 0),
    CHECK (valor_total >= 0),
    CHECK (status IN ('Pendente', 'Pago', 'Cancelado')),
    CHECK (forma_pagamento IN ('Pix', 'Cartao', 'Boleto'))
) ENGINE=InnoDB;

-- TABELA CARRINHO
CREATE TABLE carrinho (
    id_carrinho BIGINT AUTO_INCREMENT PRIMARY KEY,
    id_cliente BIGINT NOT NULL,
    valor_desconto DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    valor_frete DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    subtotal DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    quantidade_itens INT NOT NULL DEFAULT 0,
    valor_total DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    status VARCHAR(20) NOT NULL,

    CONSTRAINT fk_carrinho_cliente
        FOREIGN KEY (id_cliente)
        REFERENCES cliente(id_cliente),

    CHECK (valor_desconto >= 0),
    CHECK (valor_frete >= 0),
    CHECK (subtotal >= 0),
    CHECK (quantidade_itens >= 0),
    CHECK (valor_total >= 0),
    CHECK (status IN ('Pendente', 'Pago', 'Cancelado'))
) ENGINE=InnoDB;

-- TABELA ITEM_CARRINHO
CREATE TABLE item_carrinho (
    id_item_carrinho BIGINT AUTO_INCREMENT PRIMARY KEY,
    id_carrinho BIGINT NOT NULL,
    id_produto BIGINT NOT NULL,
    quantidade INT NOT NULL,
    preco_unitario DECIMAL(10,2) NOT NULL,
    valor_desconto DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    desconto_unitario DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    valor_frete DECIMAL(10,2) NOT NULL DEFAULT 0.00,

    CONSTRAINT fk_item_carrinho_carrinho
        FOREIGN KEY (id_carrinho)
        REFERENCES carrinho(id_carrinho),

    CONSTRAINT fk_item_carrinho_produto
        FOREIGN KEY (id_produto)
        REFERENCES produto(id_produto),

    CHECK (quantidade > 0),
    CHECK (preco_unitario >= 0),
    CHECK (valor_desconto >= 0),
    CHECK (desconto_unitario >= 0),
    CHECK (valor_frete >= 0)
) ENGINE=InnoDB;

-- TABELA ITEM_PEDIDO
CREATE TABLE item_pedido (
    id_item_pedido BIGINT AUTO_INCREMENT PRIMARY KEY,
    id_pedido BIGINT NOT NULL,
    id_produto BIGINT NOT NULL,
    quantidade INT NOT NULL,
    preco_unitario DECIMAL(10,2) NOT NULL,

    CONSTRAINT fk_item_pedido_pedido
        FOREIGN KEY (id_pedido)
        REFERENCES pedido(id_pedido),

    CONSTRAINT fk_item_pedido_produto
        FOREIGN KEY (id_produto)
        REFERENCES produto(id_produto),

    CHECK (quantidade > 0),
    CHECK (preco_unitario >= 0)
) ENGINE=InnoDB;

-- TABELA LOG_AUDITORIA
CREATE TABLE log_auditoria (
    id_log_auditoria BIGINT AUTO_INCREMENT PRIMARY KEY,
    id_cliente BIGINT NULL,
    acao VARCHAR(50) NOT NULL,
    entidade_afetada VARCHAR(100) NOT NULL,
    id_registro_afetado BIGINT NOT NULL,
    detalhe_operacao TEXT NULL,
    data_evento DATETIME NOT NULL,

    CONSTRAINT fk_log_cliente
        FOREIGN KEY (id_cliente)
        REFERENCES cliente(id_cliente),

    CHECK (acao IN ('LOGIN', 'CANCELAMENTO', 'UPDATE'))
) ENGINE=InnoDB;
